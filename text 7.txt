local function snapshot(part)
    return {
        Size = part.Size, Transparency = part.Transparency, CanCollide = part.CanCollide,
        Material = part.Material, Color = part.Color, Reflectance = part.Reflectance,
    }
end

local function untrack(part)
    local rec = balls[part]
    if not rec then return end
    if rec.conn then rec.conn:Disconnect() end
    if rec.fake then rec.fake:Destroy() end
    balls[part] = nil
    sizeWatch[part] = nil
end

local function track(part)
    if not part:IsA("BasePart") or balls[part] then return end
    local rec = { base = nil, conn = nil, fake = nil, locked = false }
    rec.conn = part.AncestryChanged:Connect(function(_, parent)
        if not parent then untrack(part) end
    end)
    balls[part] = rec
end

local function forceRecapture(part)
    local rec = balls[part]
    if not rec then return end
    rec.base = nil
    rec.locked = false
    if rec.fake then rec.fake:Destroy() rec.fake = nil end
end

local function ensureBase(part, rec)
    local liveSize = part.Size

    local valid = liveSize.X > 0.5 and liveSize.Y > 0.5 and liveSize.Z > 0.5
    if not valid then
        rec.locked = false
        rec.base = nil
        if rec.fake then rec.fake:Destroy() rec.fake = nil end
        return
    end

    if not rec.base or not rec.locked then
        rec.base = snapshot(part)
        rec.locked = true
        if rec.fake then rec.fake:Destroy() rec.fake = nil end
    end

    if not rec.fake and rec.base then
        local prev = part.Archivable
        part.Archivable = true
        local fake = part:Clone()
        part.Archivable = prev
        if not fake then return end

        for _, v in ipairs(fake:GetDescendants()) do
            if v:IsA("LuaSourceContainer") or v:IsA("BodyMover") or v:IsA("Constraint")
            or v:IsA("JointInstance") or v:IsA("TouchTransmitter") then
                v:Destroy()
            end
        end

        fake.Name = "VisualFake_" .. part.Name
        fake.Anchored = true
        fake.CanCollide = false
        fake.Massless = true
        pcall(function() fake.CanQuery = false end)
        pcall(function() fake.CanTouch = false end)
        fake.Size = rec.base.Size
        fake.Transparency = rec.base.Transparency
        fake.Material = rec.base.Material
        fake.Color = rec.base.Color
        if rec.base.Reflectance then fake.Reflectance = rec.base.Reflectance end

        rec.fake = fake
        fake.Parent = part.Parent
    end
end

local function restore(part, rec)
    if not rec.base then return end
    part.Size = rec.base.Size
    part.Transparency = rec.base.Transparency
    part.CanCollide = rec.base.CanCollide
    part.Material = rec.base.Material
    if rec.fake then rec.fake:Destroy() rec.fake = nil end
    rec.base = nil
    rec.locked = false
end

                end
            end
        end
    end

    if not enabled then return end

    local visT = hitboxAlpha

    for part, rec in pairs(balls) do

            end
        end
    end
end)

local air = { on = false, radius = 13, height = 1.7, strength = 35, damping = 0.5, conn = nil, bv = nil }

local function stopAir()
    if air.conn then air.conn:Disconnect() air.conn = nil end
    if air.bv then air.bv:Destroy() air.bv = nil end
end

local function startAir()
    stopAir()
    if not air.on then return end
    local bv = Instance.new("BodyVelocity")
    bv.MaxForce = V3(0, 9e9, 0)
    bv.Velocity = V3zero
    air.bv = bv
    air.conn = RunService.Heartbeat:Connect(function()
        local root = getRoot(lp.Character)
        local ball = getBall()
        if not (root and ball and air.bv) then return end
        if (root.Position - ball.Position).Magnitude <= air.radius then
            if air.bv.Parent ~= root then air.bv.Parent = root end
            local targetY = ball.Position.Y + air.height
            local newY = (targetY - root.Position.Y) * air.strength - root.AssemblyLinearVelocity.Y * air.damping
            air.bv.Velocity = V3(0, newY, 0)
        else
            air.bv.Velocity = V3zero
            air.bv.Parent = nil
        end
    end)
end

local function press(kc, down)
    if BLOCKED[kc] then return end
    pcall(function() VIM:SendKeyEvent(down, kc, false, game) end)
    if CFG.RawKeys and VK[kc] then
        pcall(function()
            local fn = down and keypress or keyrelease
            if fn then fn(VK[kc]) end
        end)
    end
    if CFG.SignalKeys and typeof(getconnections) == "function" then
        pcall(function()
            local fake = {
                KeyCode = kc,
                UserInputType = Enum.UserInputType.Keyboard,
                UserInputState = down and Enum.UserInputState.Begin or Enum.UserInputState.End,
            }
            local sig = down and UIS.InputBegan or UIS.InputEnded
            for _, c in ipairs(getconnections(sig)) do
                pcall(function() c:Fire(fake, false) end)
            end
        end)
    end
end

local function tap(kc) press(kc, true) press(kc, false) end

local function jumpOnce()
    press(K.JUMP, true)
    pcall(function()
        local h = myHum()
        if h and h.Parent and h.Health > 0 then
            h:SetStateEnabled(Enum.HumanoidStateType.Jumping, true)
            h:SetStateEnabled(Enum.HumanoidStateType.Freefall, true)
            h.Jump = true
            h:ChangeState(Enum.HumanoidStateType.Jumping)
        end
    end)
    task.wait(CFG.JumpHold)
    press(K.JUMP, false)
    task.wait(CFG.JumpLead)
end

local function hold(kc)
    heldKeys[kc] = os.clock() + CFG.MaxHold
    press(kc, true)
end

local function releaseAll()
    for kc in pairs(heldKeys) do heldKeys[kc] = nil end
    for _, kc in ipairs(PRESSABLE) do press(kc, false) end
    for _, kc in ipairs(CLEAN) do
        pcall(function() VIM:SendKeyEvent(false, kc, false, game) end)
        if keyrelease and CLEAN_VK[kc] then pcall(function() keyrelease(CLEAN_VK[kc]) end) end
    end
    local h = myHum()
    if h and h.Parent and h.Health > 0 then
        pcall(function() h:Move(Vector3.zero, false) end)
    end
end

        end
    end
end)

local function camRight()
    local root = getRoot(lp.Character)
    if not root then return Vector3.new(1, 0, 0) end
    local cam = workspace.CurrentCamera
    local r = (cam and cam.CFrame or root.CFrame).RightVector
    local flat = Vector3.new(r.X, 0, r.Z)
    if flat.Magnitude < 0.05 then
        r = root.CFrame.RightVector
        flat = Vector3.new(r.X, 0, r.Z)
    end
    if flat.Magnitude < 0.05 then return Vector3.new(1, 0, 0) end
    return flat.Unit
end

local function anchorBrief(duration)
    local root = getRoot(lp.Character)
    if not root or not root.Parent then return end
    for _, c in ipairs(root:GetChildren()) do
        if c.Name == "OutlawCounter" then c:Destroy() end
    end
    local prev = root.Anchored
    root.Anchored = true
    local bv = Instance.new("BodyVelocity")
    bv.Name = "OutlawCounter"
    bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
    bv.P = 1250
    bv.Velocity = Vector3.zero
    bv.Parent = root
    task.delay(duration, function()
        if root and root.Parent then
            root.Anchored = prev
            root.AssemblyLinearVelocity = Vector3.zero
            root.AssemblyAngularVelocity = Vector3.zero
        end
        if bv and bv.Parent then bv:Destroy() end
    end)
end

local function dive(sideSign, band)
    if not pcEnabled and not mobEnabled then return end
    if diving then return end
    diving = true

    local hum = myHum()
    if not hum or not hum.Parent or hum.Health <= 0 then
        diving = false
        return
    end

    local isHigh = band == "MID" or band == "SUPER"
    local diveKey = isHigh and K.HIGH or K.LOW
    local catchKey = isHigh and K.HCAT or K.LCAT
    local isSuper = band == "SUPER"

    local sideKey = nil
    if sideSign > 0 then sideKey = K.RIGHT
    elseif sideSign < 0 then sideKey = K.LEFT end

    local moveVec = nil
    if sideSign ~= 0 then
        moveVec = camRight() * sideSign
        pcall(function() hum:Move(moveVec, false) end)
    end
    if sideKey then hold(sideKey) end

    task.spawn(function()
        local caught = false
        local conn = nil

        local ok, err = pcall(function()
            if isSuper then
                jumpOnce()
                if not hum.Parent or hum.Health <= 0 then return end
            end

            conn = hum.Touched:Connect(function(hit)
                if caught then return end
                if hit and hit.Name == BALL then
                    caught = true
                    pcall(function() hum:Move(Vector3.zero, false) end)
                    anchorBrief(CFG.CatchAnchorTime)
                end
            end)

            tap(diveKey)
            if CFG.DoubleTap then
                tap(catchKey)
                tap(diveKey)
                tap(catchKey)
            end

            local t0 = os.clock()
            while os.clock() - t0 < CFG.HoldAfter do
                if caught then break end
                if not hum.Parent or hum.Health <= 0 then break end
                if moveVec then pcall(function() hum:Move(moveVec, false) end) end
                task.wait(0.02)
            end
        end)

        if conn then conn:Disconnect() end
        if not ok then warn("[Outlaw] dive: " .. tostring(err)) end

        pcall(function()
            local h = getRoot(lp.Character)
            if h and h.Parent and not h.Anchored then
                h.AssemblyLinearVelocity = Vector3.zero
                h.AssemblyAngularVelocity = Vector3.zero
            end
        end)
        pcall(releaseAll)
        diving = false
    end)
end

local btnSizes, btnPos, btnRefs = {}, {}, {}

local function keyName(kc) return kc and kc.Name or "None" end

local function nameKey(nm)
    if not nm or nm == "None" then return nil end
    local ok, kc = pcall(function() return Enum.KeyCode[nm] end)
    return ok and kc or nil
end

local function buildPayload()
    local posOut = {}
    for id, ud in pairs(btnPos) do
        posOut[id] = { xs = ud.X.Scale, xo = ud.X.Offset, ys = ud.Y.Scale, yo = ud.Y.Offset }
    end
    local keysOut = {}
    for slot, kc in pairs(PCKeys) do keysOut[slot] = keyName(kc) end
    return {
        v = 2,
        sizes = btnSizes,
        positions = posOut,
        cfg = { size = btnCfg.Size, opacity = btnCfg.Opacity, locked = btnCfg.Locked },
        pckeys = keysOut,
    }
end

local function applyPayload(data)
    if type(data) ~= "table" then return false end
    if type(data.sizes) == "table" then
        for k, v in pairs(data.sizes) do
            if type(v) == "number" then btnSizes[k] = v end
        end
    end
    if type(data.positions) == "table" then
        for k, v in pairs(data.positions) do
            if type(v) == "table" and v.xs and v.xo and v.ys and v.yo then
                btnPos[k] = UDim2.new(v.xs, v.xo, v.ys, v.yo)
            end
        end
    end
    if type(data.cfg) == "table" then
        if type(data.cfg.size) == "number" then btnCfg.Size = data.cfg.size end
        if type(data.cfg.opacity) == "number" then btnCfg.Opacity = data.cfg.opacity end
        if type(data.cfg.locked) == "boolean" then btnCfg.Locked = data.cfg.locked end
    end
    if type(data.pckeys) == "table" then
        for slot, nm in pairs(data.pckeys) do
            local kc = nameKey(nm)
            if kc then PCKeys[slot] = kc end
        end
    end
    return true
end

local function saveLayout()
    if not canSave then return end
    pcall(function() writefile(SAVE_FILE, Http:JSONEncode(buildPayload())) end)
end

local function loadLayout()
    if not canSave then return end
    local exists = false
    pcall(function() exists = isfile(SAVE_FILE) end)
    if not exists then return end
    local data
    local ok = pcall(function() data = Http:JSONDecode(readfile(SAVE_FILE)) end)
    if ok then applyPayload(data) end
end

            if mobEnabled then dive(def.side, def.band) end
            fired = true
            return
        end

        startPos = btn.Position
        fired = false
    end)

        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch then
            local d = input.Position - dragStart
            btn.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + d.X,
                startPos.Y.Scale, startPos.Y.Offset + d.Y
            )
        end
    end)

        and input.UserInputType ~= Enum.UserInputType.Touch then return end
        if dragging then
            dragging = false
            btnPos[def.id] = btn.Position
            saveLayout()
            if (input.Position - (dragStart or input.Position)).Magnitude < 6 and not fired then
                if mobEnabled then dive(def.side, def.band) end
            end
        end
        fired = false
    end)

    btnRefs[def.id] = { btn = btn, corner = corner, stroke = stroke, def = def }
end

local function probe(url)
    local ok, res = pcall(function() return game:HttpGet(url) end)
    if not ok then return false, "HttpGet failed: " .. tostring(res) end
    if type(res) ~= "string" then return false, "HttpGet returned " .. typeof(res) end
    if #res == 0 then return false, "empty body" end
    local head = res:sub(1, 200):lower()
    if head:find("<!doctype") or head:find("<html") or head:find("<head") then
        return false, "URL returned HTML not Lua:\n\n" .. res:sub(1, 200)
    end
    return true, res
end

local function loadCustom()
    if customLoaded then return true end
    local ok, body = probe(customURL)
    if not ok then
        warn("[Outlaw] custom controls: " .. body)
        showPopup("Custom Controls: fetch failed", body)
        return false
    end

    local ls = loadstring or load
    if type(ls) ~= "function" then
        showPopup("Custom Controls: execute failed", "no loadstring/load")
        return false
    end

    local fn, perr = ls(body)
    if not fn then
        local msg = "compile error from " .. customURL .. ":\n\n" .. tostring(perr)
        warn("[Outlaw] custom controls: " .. msg)
        showPopup("Custom Controls: execute failed", msg)
        return false
    end

    local ran, rerr = pcall(fn)
    if not ran then
        local msg = "runtime error from " .. customURL .. ":\n\n" .. tostring(rerr)
        warn("[Outlaw] custom controls: " .. msg)
        showPopup("Custom Controls: execute failed", msg)
        return false
    end

    customLoaded = true
    return true
end
